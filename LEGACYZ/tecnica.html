Pantalla LCD I2C

4 cables

VCC5V

El módulo I2C reduce las conexiones de la pantalla de 16 a 4. Dirección del módulo: 0x27.

Teclado matricial 4×4

8 cables
S1 · S2 · S3 · S4 (filas)9 · 8 · 7 · 6
S5 · S6 · S7 · S8 (columnas)5 · 4 · 3 · 2
S9 a S16sin uso

Los pines S9 a S16 del módulo están duplicados internamente: con los ocho primeros alcanza.

Librerías necesarias

Se instalan desde el Gestor de Librerías del IDE de Arduino antes de compilar.

Wire.h

Comunicación I2C. Viene incluida con el IDE.

LiquidCrystal_I2C.h

Control de la pantalla a través del módulo I2C.

Keypad.h

Lectura del teclado matricial por barrido de filas y columnas.

Software

Código completo del Arduino

Versión definitiva, con los textos ajustados a las 16 columnas de la pantalla y el límite de tres dígitos en el peso.

calculadora_proteina.ino C++ · Arduino 
#include <Wire.h>
#include <LiquidCrystal_I2C.h>
#include <Keypad.h>

// Pantalla LCD I2C — direccion 0x27, 16 columnas x 2 filas
LiquidCrystal_I2C lcd(0x27, 16, 2);

// ----- Teclado matricial 4x4 -----
const byte FILAS = 4;
const byte COLUMNAS = 4;

char matriz[FILAS][COLUMNAS] = {
  {'1','2','3','A'},
  {'4','5','6','B'},
  {'7','8','9','C'},
  {'*','0','#','D'}
};

byte pinesFilas[FILAS]       = {9, 8, 7, 6};   // S1 a S4
byte pinesColumnas[COLUMNAS] = {5, 4, 3, 2};   // S5 a S8

Keypad teclado = Keypad(makeKeymap(matriz), pinesFilas, pinesColumnas, FILAS, COLUMNAS);

// ----- Estados del programa -----
enum Estados { SELECCION_MENU, INGRESO_PESO, MOSTRAR_RESULTADO };
Estados estadoActual = SELECCION_MENU;

int    opcionSeleccionada = 0;
float  factorProteina     = 0.0;
String stringPeso         = "";
int    pesoIngresado      = 0;

void setup() {
  lcd.init();
  lcd.backlight();
  mostrarMenuInicial();
}

void loop() {
  char tecla = teclado.getKey();
  if (!tecla) return;

  // La tecla C reinicia el sistema en cualquier estado
  if (tecla == 'C') { reiniciarCalculadora(); return; }

  switch (estadoActual) {

    case SELECCION_MENU:
      if (tecla >= '1' && tecla <= '3') {
        opcionSeleccionada = tecla - '0';
        if      (opcionSeleccionada == 1) factorProteina = 0.9;  // Sedentario
        else if (opcionSeleccionada == 2) factorProteina = 1.9;  // Atleta
        else                              factorProteina = 2.1;  // Perdida de grasa
        estadoActual = INGRESO_PESO;
        pantallaIngresoPeso();
      }
      break;

    case INGRESO_PESO:
      // Hasta 3 digitos: asi el texto nunca desborda las 16 columnas
      if (tecla >= '0' && tecla <= '9' && stringPeso.length() < 3) {
        stringPeso += tecla;
        lcd.setCursor(5 + stringPeso.length() - 1, 1);
        lcd.print(tecla);
      }
      else if (tecla == 'A' && stringPeso.length() > 0) {   // A = Enter
        pesoIngresado = stringPeso.toInt();
        estadoActual  = MOSTRAR_RESULTADO;
        calcularYMostrarResultado();
      }
      break;

    case MOSTRAR_RESULTADO:
      reiniciarCalculadora();   // cualquier tecla vuelve al menu
      break;
  }
}

// ----- Pantallas -----

void mostrarMenuInicial() {
  lcd.clear();
  lcd.setCursor(0, 0);
  lcd.print("1Sed 2Atl 3Grasa");   // 16 caracteres exactos
  lcd.setCursor(0, 1);
  lcd.print("Elija opcion...");
}

void pantallaIngresoPeso() {
  lcd.clear();
  lcd.setCursor(0, 0);
  if      (opcionSeleccionada == 1) lcd.print("Modo: Sedentario");
  else if (opcionSeleccionada == 2) lcd.print("Modo: Atleta");
  else                              lcd.print("Modo: Deficit");
  lcd.setCursor(0, 1);
  lcd.print("Peso:___kg  A=OK");
}

void calcularYMostrarResultado() {
  float gramos = pesoIngresado * factorProteina;   // Peso x Factor
  lcd.clear();
  lcd.setCursor(0, 0);
  lcd.print("Proteina Diaria:");
  lcd.setCursor(0, 1);
  lcd.print(gramos, 1);
  lcd.print(" g al dia");
}

void reiniciarCalculadora() {
  stringPeso         = "";
  pesoIngresado      = 0;
  factorProteina     = 0.0;
  opcionSeleccionada = 0;
  estadoActual       = SELECCION_MENU;
  mostrarMenuInicial();
}

Decisiones técnicas

  • Punto medio en lugar de rango. Cada categoría tiene un intervalo de gramos por kilo. El dispositivo usa el punto medio para devolver un número único; la calculadora web muestra además el mínimo y el máximo.
  • Tres opciones en el dispositivo, cinco en la web. En 16 columnas no caben cinco abreviaturas legibles. Las categorías Activo y Adulto mayor están disponibles en la versión web.
  • Peso de hasta tres dígitos. Cubre cualquier peso corporal real y garantiza que el texto nunca desborde la pantalla.
  • Un decimal en el resultado. lcd.print(gramos, 1) evita una cifra larga y mantiene la precisión suficiente.
